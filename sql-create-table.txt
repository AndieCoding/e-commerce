-----triggers-----
---- triggers para ficha de stock hacer uno para cada tabla de las fichas

CREATE DEFINER=`root`@`localhost` TRIGGER `ficha_stock_mates_BEFORE_INSERT` BEFORE INSERT ON `ficha_stock_mates` FOR EACH ROW BEGIN
 IF NEW.TIPO = 'duplicado' THEN
        UPDATE productos
        SET P_CANTIDAD = P_CANTIDAD - NEW.V_CANTIDAD
        WHERE productos.p_precio = NEW.v_precio; 
    END IF;
END
